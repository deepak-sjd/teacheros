    tenant_id    UUID NOT NULL REFERENCES tenants(id),
    structure_id UUID NOT NULL REFERENCES fee_structures(id),
    head_id      UUID NOT NULL REFERENCES fee_heads(id),
    amount       NUMERIC(10, 2) NOT NULL CHECK (amount > 0),
    CONSTRAINT uq_fee_structure_items UNIQUE (structure_id, head_id)
);
CREATE INDEX idx_fee_structure_items_structure ON fee_structure_items(tenant_id, structure_id);

-- Additive changes to the existing fees table. Every new column is nullable
-- or has a default, so existing rows and the existing Fee/Payment code are
-- unaffected (line_type defaults to STANDARD, discount_amount to 0).
--   fee_head_id    set only when the structure has exactly one head
--   gross_amount   amount before discounts (amount_due = gross - discount)
--   parent_fee_id  for LATE_FEE rows: the overdue instalment charged
ALTER TABLE fees
    ADD COLUMN fee_head_id     UUID REFERENCES fee_heads(id),
    ADD COLUMN structure_id    UUID REFERENCES fee_structures(id),
    ADD COLUMN instalment_no   INT,
    ADD COLUMN line_type       VARCHAR(20) NOT NULL DEFAULT 'STANDARD',
    ADD COLUMN gross_amount    NUMERIC(10, 2),
    ADD COLUMN discount_amount NUMERIC(10, 2) NOT NULL DEFAULT 0,
    ADD COLUMN parent_fee_id   UUID REFERENCES fees(id);

-- Idempotency: one fee per (student, structure, instalment).
CREATE UNIQUE INDEX uq_fees_structure_instalment
    ON fees (tenant_id, student_id, structure_id, instalment_no)
    WHERE line_type = 'INSTALMENT' AND structure_id IS NOT NULL;

-- Late fee is charged at most once per overdue instalment.
CREATE UNIQUE INDEX uq_fees_late_fee_parent
    ON fees (parent_fee_id)
    WHERE line_type = 'LATE_FEE';

CREATE INDEX idx_fees_tenant_structure ON fees (tenant_id, structure_id) WHERE structure_id IS NOT NULL;
CREATE INDEX idx_fees_tenant_due ON fees (tenant_id, due_date);

-- Per-head breakdown of a generated fee. Payments are recorded against the
-- fee (not the head), so the "collection per fee head" report allocates each
-- payment across these lines pro rata to (gross - discount).
CREATE TABLE fee_line_items (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id       UUID NOT NULL REFERENCES tenants(id),
    fee_id          UUID NOT NULL REFERENCES fees(id),
    head_id         UUID NOT NULL REFERENCES fee_heads(id),
    gross_amount    NUMERIC(10, 2) NOT NULL,
    discount_amount NUMERIC(10, 2) NOT NULL DEFAULT 0,
    CONSTRAINT uq_fee_line_items UNIQUE (fee_id, head_id)
);
CREATE INDEX idx_fee_line_items_fee ON fee_line_items(tenant_id, fee_id);

package com.teacheros.modules.tenant;

import jakarta.persistence.*;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;
import org.hibernate.annotations.JdbcTypeCode;
import org.hibernate.type.SqlTypes;

import java.time.Instant;
import java.util.ArrayList;
import java.util.List;
import java.util.UUID;

@Entity
@Table(name = "tenant_config")
@Getter
@Setter
@NoArgsConstructor
public class TenantConfig {

    @Id
    @GeneratedValue
    private UUID id;

    @Column(name = "tenant_id", nullable = false, updatable = false, unique = true)
    private UUID tenantId;

    @Enumerated(EnumType.STRING)
    @Column(name = "tenant_type", nullable = false, length = 20)
    private TenantType tenantType = TenantType.COACHING;

    @JdbcTypeCode(SqlTypes.JSON)
    @Column(name = "enabled_modules", nullable = false, columnDefinition = "jsonb")
    private List<String> enabledModules = new ArrayList<>();

    @Column(nullable = false, length = 20)
    private String locale = "en-IN";

    @Column(nullable = false, length = 50)
    private String timezone = "Asia/Kolkata";

    @Column(name = "primary_color", length = 7)
    private String primaryColor;

    @Column(name = "secondary_color", length = 7)
    private String secondaryColor;

    @Column(name = "created_at", nullable = false, updatable = false)
    private Instant createdAt;

    @Column(name = "updated_at", nullable = false)
    private Instant updatedAt;

    @PrePersist
    void onCreate() {
        Instant now = Instant.now();
        this.createdAt = now;
        this.updatedAt = now;
    }

    @PreUpdate
    void onUpdate() {
        this.updatedAt = Instant.now();
    }

    public TenantConfig(UUID tenantId, TenantType tenantType, List<String> enabledModules) {
        this.tenantId = tenantId;
        this.tenantType = tenantType;
        this.enabledModules = new ArrayList<>(enabledModules);
    }

package com.teacheros.modules.tenant;

import com.teacheros.common.exception.ApiException;
import com.teacheros.common.tenant.CurrentTenantProvider;
import com.teacheros.modules.tenant.dto.TenantDtos.TenantConfigResponse;
import com.teacheros.modules.tenant.dto.TenantDtos.UpdateTenantConfigRequest;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.DateTimeException;
import java.time.ZoneId;
import java.util.ArrayList;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.UUID;

/**
 * The one place other modules read tenant type / enabled modules from:
 * TenantConfigService#get() for the current request's tenant,
 * #isModuleEnabled(key) for a quick check, #getFor(tenantId) for
 * background jobs that have no request context.
 */
@Service
public class TenantConfigService {

    private final TenantConfigRepository tenantConfigRepository;
    private final CurrentTenantProvider currentTenantProvider;

    public TenantConfigService(TenantConfigRepository tenantConfigRepository,
            CurrentTenantProvider currentTenantProvider) {
        this.tenantConfigRepository = tenantConfigRepository;
        this.currentTenantProvider = currentTenantProvider;
    }

    @Transactional
    public TenantConfigResponse get() {
        return toResponse(getOrCreate(currentTenantProvider.getCurrentTenantId()));
    }

    /**
     * For scheduled jobs only - request code must use get(), never a
     * client-supplied id.
     */
    @Transactional
    public TenantConfigResponse getFor(UUID tenantId) {
        return toResponse(getOrCreate(tenantId));
    }

    public boolean isModuleEnabled(String moduleKey) {
        return get().enabledModules().contains(moduleKey);
    }

    public ZoneId zone() {
        return ZoneId.of(get().timezone());
    }

    /**
     * Called at signup (AuthService.provisionNewTeacherAccount) inside the
     * same transaction that creates the tenant, so a tenant never exists
     * without a config row. Takes the id explicitly because there's no
     * authenticated principal yet at that point.
     */
    @Transactional
    public TenantConfigResponse createDefault(UUID tenantId) {
        return toResponse(tenantConfigRepository.save(newDefault(tenantId)));
    }

    @Transactional
    public TenantConfigResponse update(UpdateTenantConfigRequest request) {
        TenantConfig config = getOrCreate(currentTenantProvider.getCurrentTenantId());

        if (request.tenantType() != null) {
            config.setTenantType(parseType(request.tenantType()));
        }
        if (request.enabledModules() != null) {
            config.setEnabledModules(validateModules(request.enabledModules()));
        }
        if (request.locale() != null) {
            config.setLocale(request.locale());
        }
        if (request.timezone() != null) {
            config.setTimezone(validateZone(request.timezone()));
        }
        // null = unchanged, "" = clear
        if (request.primaryColor() != null) {
            config.setPrimaryColor(blankToNull(request.primaryColor()));
        }
        if (request.secondaryColor() != null) {
            config.setSecondaryColor(blankToNull(request.secondaryColor()));
        }
        return toResponse(tenantConfigRepository.save(config));
    }

    // Self-heal: the V20 backfill and signup hook mean a row should always
    // exist, but a read must never fail because one is missing.
    private TenantConfig getOrCreate(UUID tenantId) {
        return tenantConfigRepository.findByTenantId(tenantId)
                .orElseGet(() -> tenantConfigRepository.save(newDefault(tenantId)));
    }

    private TenantConfig newDefault(UUID tenantId) {
        return new TenantConfig(tenantId, TenantType.COACHING, TenantModules.defaultsFor(TenantType.COACHING));
    }

    private TenantType parseType(String value) {
        try {
            return TenantType.valueOf(value.trim().toUpperCase());
        } catch (IllegalArgumentException e) {
            throw new ApiException(HttpStatus.BAD_REQUEST, "INVALID_TENANT_TYPE",
                    "tenantType must be COACHING or SCHOOL");
        }
    }

    private List<String> validateModules(List<String> modules) {
        LinkedHashSet<String> cleaned = new LinkedHashSet<>();
        for (String m : modules) {
            String key = m == null ? "" : m.trim().toUpperCase();
            if (!TenantModules.ALL.contains(key)) {
                throw new ApiException(HttpStatus.BAD_REQUEST, "UNKNOWN_MODULE", "Unknown module: " + m);
            }
            cleaned.add(key);
        }
        return new ArrayList<>(cleaned);
    }

    private String validateZone(String zone) {
        try {
            return ZoneId.of(zone.trim()).getId();
        } catch (DateTimeException e) {
            throw new ApiException(HttpStatus.BAD_REQUEST, "INVALID_TIMEZONE", "Unknown timezone: " + zone);
        }
    }

    private String blankToNull(String s) {
        return s.isBlank() ? null : s.trim();
    }

    private TenantConfigResponse toResponse(TenantConfig c) {
        return new TenantConfigResponse(c.getTenantType(), List.copyOf(c.getEnabledModules()), c.getLocale(),
                c.getTimezone(), c.getPrimaryColor(), c.getSecondaryColor());
    }
}

package com.teacheros.modules.tenant;

import java.util.LinkedHashSet;
import java.util.List;
import java.util.Set;

/**
 * The contract for module keys stored in tenant_config.enabled_modules.
 * A dev adding a new toggleable module appends its key to ALL in the same
 * PR as the feature. Unknown keys are rejected on PATCH so a typo can't
 * silently disable something.
 */
public final class TenantModules {

    private TenantModules() {
    }

    public static final Set<String> ALL = Set.copyOf(new LinkedHashSet<>(List.of(
            "ATTENDANCE", "FEES", "HOMEWORK", "TIMETABLE", "STUDY_MATERIAL", "TESTS",
            "ALERTS", "MESSAGES", "DOUBTS", "REPORTS", "ANALYTICS")));

    // Same set for both types today; SCHOOL-only modules get added by
    // whoever builds them.
    public static List<String> defaultsFor(TenantType type) {
        return List.of("ATTENDANCE", "FEES", "HOMEWORK", "TIMETABLE", "STUDY_MATERIAL", "TESTS",
                "ALERTS", "MESSAGES", "DOUBTS", "REPORTS", "ANALYTICS");
    }
}

package com.teacheros.modules.student;

import com.teacheros.common.exception.ApiException;
import com.teacheros.common.exception.ResourceNotFoundException;
import com.teacheros.common.tenant.CurrentTenantProvider;
import com.teacheros.modules.audit.AuditActions;
import com.teacheros.modules.audit.AuditService;
import com.teacheros.modules.batch.BatchRepository;
import com.teacheros.modules.student.dto.StudentDtos.CreateStudentRequest;
import com.teacheros.modules.student.dto.StudentDtos.StudentResponse;
import com.teacheros.modules.student.dto.StudentDtos.UpdateStudentRequest;
import com.teacheros.modules.user.User;
import com.teacheros.modules.user.UserRepository;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;
import java.util.UUID;

@Service
public class StudentService {

    private static final int MAX_LOOKUP_IDS = 200;

    private final StudentRepository studentRepository;
    private final UserRepository userRepository;
    private final BatchRepository batchRepository;
    private final CurrentTenantProvider currentTenantProvider;
    private final StudentAccessGuard studentAccessGuard;
    private final AuditService auditService;

    public StudentService(
            StudentRepository studentRepository,
            UserRepository userRepository,
            BatchRepository batchRepository,
            CurrentTenantProvider currentTenantProvider,
            StudentAccessGuard studentAccessGuard,
            AuditService auditService) {
        this.studentRepository = studentRepository;
        this.userRepository = userRepository;
        this.batchRepository = batchRepository;
        this.currentTenantProvider = currentTenantProvider;
        this.studentAccessGuard = studentAccessGuard;
        this.auditService = auditService;
    }

    @Transactional
    public StudentResponse create(CreateStudentRequest request) {
        UUID tenantId = currentTenantProvider.getCurrentTenantId();
        assertBatchBelongsToTenant(tenantId, request.batchId());
        assertPhoneAvailable(tenantId, request.phoneNumber(), null);

        Student student = new Student(tenantId, request.batchId(), request.fullName(),
                request.grade(), request.guardianPhone());
        student.setPhoneNumber(request.phoneNumber());
        student.setGuardianEmail(request.guardianEmail());
        student = studentRepository.save(student);
        return toResponse(student);
    }

    /**
     * Staff-only (see controller's @PreAuthorize) - a student/parent editing
     * their own contact details isn't supported yet, this is the teacher
     * correcting/completing a record, e.g. adding the student's own phone
     * after the fact so they can log in themselves.
     */
    @Transactional
    public StudentResponse update(UUID studentId, UpdateStudentRequest request) {
        UUID tenantId = currentTenantProvider.getCurrentTenantId();
        Student student = studentRepository.findByIdAndTenantId(studentId, tenantId)
                .orElseThrow(() -> new ResourceNotFoundException("Student", studentId));
        assertBatchBelongsToTenant(tenantId, request.batchId());
        assertPhoneAvailable(tenantId, request.phoneNumber(), studentId);

        student.setFullName(request.fullName());
        student.setGrade(request.grade());
        student.setGuardianPhone(request.guardianPhone());
        student.setPhoneNumber(request.phoneNumber());
        student.setGuardianEmail(request.guardianEmail());
        student.setBatchId(request.batchId());
        student = studentRepository.save(student);
        return toResponse(student);
    }

    /**
     * Soft delete - attendance, marks, fees and messages keep referencing
     * this student, so the row stays. Inactive students disappear from
     * lists, rosters, reports, alerts, and from parent/student account
     * linking. Idempotent (and only audited when something actually changed).
     */
    @Transactional
    public void deactivate(UUID studentId) {
        UUID tenantId = currentTenantProvider.getCurrentTenantId();
        Student student = studentRepository.findByIdAndTenantId(studentId, tenantId)
                .orElseThrow(() -> new ResourceNotFoundException("Student", studentId));
        if (student.isActive()) {
            student.setActive(false);
            studentRepository.save(student);
            // Name only - phone numbers are deliberately not written to the audit log.
            auditService.record(AuditActions.STUDENT_DEACTIVATED, AuditActions.STUDENT, studentId,
                    AuditService.meta("fullName", student.getFullName(), "batchId", student.getBatchId()));
        }
    }

    /**
     * search (optional) matches a name or guardian-phone fragment,
     * case-insensitively, ordered by name. Blank/null keeps the original
     * unsearched behaviour.
     */
    public Page<StudentResponse> list(UUID batchId, String search, Pageable pageable) {
        UUID tenantId = currentTenantProvider.getCurrentTenantId();

        if (search != null && !search.isBlank()) {
            String pattern = "%" + escapeLike(search.trim().toLowerCase()) + "%";
            return studentRepository.search(tenantId, batchId, pattern, pageable).map(this::toResponse);
        }

        Page<Student> page = (batchId != null)
                ? studentRepository.findByTenantIdAndBatchIdAndActiveTrue(tenantId, batchId, pageable)
                : studentRepository.findByTenantIdAndActiveTrue(tenantId, pageable);
        return page.map(this::toResponse);
    }

    /**
     * Staff-only: resolve a set of student IDs to records (e.g. names for a page of
     * fees).
     */
    public List<StudentResponse> lookup(List<UUID> ids) {
        if (ids.size() > MAX_LOOKUP_IDS) {
            throw new ApiException(HttpStatus.BAD_REQUEST, "TOO_MANY_IDS",
                    "Look up at most " + MAX_LOOKUP_IDS + " students at a time.");
        }
        UUID tenantId = currentTenantProvider.getCurrentTenantId();
        return studentRepository.findByTenantIdAndIdIn(tenantId, ids.stream().distinct().toList())
                .stream().map(this::toResponse).toList();
    }

    /**
     * Any authenticated role can call this (see StudentController - no
     * PreAuthorize restricting it), but PARENT/STUDENT callers only ever
     * get their own linked student back - studentAccessGuard enforces that
     * before we touch anything else. Staff roles are unaffected.
     */
    public StudentResponse get(UUID studentId) {
        studentAccessGuard.assertCanView(studentId);
        UUID tenantId = currentTenantProvider.getCurrentTenantId();
        Student student = studentRepository.findByIdAndTenantId(studentId, tenantId)
                .orElseThrow(() -> new ResourceNotFoundException("Student", studentId));
        return toResponse(student);
    }

    /**
     * PARENT self-service: every active student in this tenant whose
     * guardian_phone matches the calling user's own phone - siblings
     * included, no separate linking step the teacher has to perform.
     */
    public List<StudentResponse> listMyChildren() {
        UUID tenantId = currentTenantProvider.getCurrentTenantId();
        String phoneNumber = currentUserPhoneNumber();
        return studentRepository.findByTenantIdAndGuardianPhoneAndActiveTrue(tenantId, phoneNumber)
                .stream()
                .map(this::toResponse)
                .toList();
    }

    /**
     * STUDENT self-service: the one student record whose own phone_number
     * matches the calling user's login phone.
     */
    public StudentResponse getMyStudentRecord() {
        UUID tenantId = currentTenantProvider.getCurrentTenantId();
        String phoneNumber = currentUserPhoneNumber();
        Student student = studentRepository.findFirstByTenantIdAndPhoneNumberAndActiveTrue(tenantId, phoneNumber)
                .orElseThrow(() -> new ResourceNotFoundException("Student", "self"));
        return toResponse(student);
    }

    // A client-supplied batchId must belong to the caller's own tenant -
    // without this a student could be enrolled in another tenant's batch.
    private void assertBatchBelongsToTenant(UUID tenantId, UUID batchId) {
        if (batchId == null) {
            return;
        }
        batchRepository.findByIdAndTenantId(batchId, tenantId)
                .orElseThrow(() -> new ResourceNotFoundException("Batch", batchId));
    }

    // uq_students_tenant_phone would otherwise surface as a raw 500.
    private void assertPhoneAvailable(UUID tenantId, String phoneNumber, UUID selfId) {
        if (phoneNumber == null || phoneNumber.isBlank()) {
            return;
        }
        studentRepository.findFirstByTenantIdAndPhoneNumber(tenantId, phoneNumber)
                .filter(other -> !other.getId().equals(selfId))
                .ifPresent(other -> {
                    throw new ApiException(HttpStatus.CONFLICT, "STUDENT_PHONE_IN_USE",
                            "Another student (possibly a deactivated one) already uses that phone number.");
                });
    }

    private String currentUserPhoneNumber() {
        User user = userRepository.findById(currentTenantProvider.getCurrentUserId())
                .orElseThrow(
                        () -> new IllegalStateException("Authenticated user not found - this should never happen"));
        return user.getPhoneNumber();
    }

    // '!' is the ESCAPE character in StudentRepository#search.
    private String escapeLike(String value) {
        return value.replace("!", "!!").replace("%", "!%").replace("_", "!_");
    }

    private StudentResponse toResponse(Student student) {
        return new StudentResponse(student.getId(), student.getFullName(), student.getGrade(),
                student.getGuardianPhone(), student.getPhoneNumber(), student.getGuardianEmail(),
                student.getBatchId());
    }
}

package com.teacheros.modules.student;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

import java.util.Collection;
import java.util.List;
import java.util.Optional;
import java.util.UUID;

public interface StudentRepository extends JpaRepository<Student, UUID> {

    Optional<Student> findByIdAndTenantId(UUID id, UUID tenantId);

    Page<Student> findByTenantIdAndActiveTrue(UUID tenantId, Pageable pageable);

    /**
     * Name / guardian-phone search for the student picker. pattern is a
     * ready-made lowercase LIKE pattern (see StudentService#list) using
     * '!' as the escape character. Ordering is fixed here, so callers
     * must not also pass a sort.
     */
    @Query("""
            SELECT s FROM Student s
            WHERE s.tenantId = :tenantId AND s.active = true
            AND (:batchId IS NULL OR s.batchId = :batchId)
            AND (LOWER(s.fullName) LIKE :pattern ESCAPE '!' OR s.guardianPhone LIKE :pattern ESCAPE '!')
            ORDER BY s.fullName
            """)
    Page<Student> search(@Param("tenantId") UUID tenantId, @Param("batchId") UUID batchId,
            @Param("pattern") String pattern, Pageable pageable);

    /**
     * Resolve names for a set of IDs (e.g. the students behind a page of
     * fees). Includes inactive students - a fee outlives a deactivation.
     */
    List<Student> findByTenantIdAndIdIn(UUID tenantId, Collection<UUID> ids);

    /**
     * Unpaged variant used by NotificationService#generateLowAttendanceAlerts -
     * it has to walk every active student in the tenant to compute each
     * one's attendance percentage, so pagination doesn't apply there.
     */
    List<Student> findAllByTenantIdAndActiveTrue(UUID tenantId);

    Page<Student> findByTenantIdAndBatchIdAndActiveTrue(UUID tenantId, UUID batchId, Pageable pageable);

    List<Student> findByTenantIdAndBatchIdAndActiveTrue(UUID tenantId, UUID batchId);

    /**
     * Used by BatchService to refuse deactivating a batch that still has students.
     */
    long countByTenantIdAndBatchIdAndActiveTrue(UUID tenantId, UUID batchId);

    /**
     * All active students in a tenant whose guardian_phone matches - i.e.
     * this guardian's children, including siblings. This is how a parent
     * account's linked children are derived: no separate parent-student
     * link table, just a query against the phone already collected when
     * the student was added.
     */
    List<Student> findByTenantIdAndGuardianPhoneAndActiveTrue(UUID tenantId, String guardianPhone);

    /**
     * The student whose own phone_number matches, scoped to a tenant - used
     * once we already know which tenant the logged-in user belongs to.
     */
    Optional<Student> findFirstByTenantIdAndPhoneNumberAndActiveTrue(UUID tenantId, String phoneNumber);

    /**
     * Active OR inactive - the unique index uq_students_tenant_phone covers
     * both, so duplicate-phone checks on create/update must too.
     */
    Optional<Student> findFirstByTenantIdAndPhoneNumber(UUID tenantId, String phoneNumber);

    /**
     * Every student phone_number in the tenant, active or not - the unique
     * index uq_students_tenant_phone applies to all rows, so StudentImportService
     * must check against inactive students' numbers too.
     */
    @Query("SELECT s.phoneNumber FROM Student s WHERE s.tenantId = :tenantId AND s.phoneNumber IS NOT NULL")
    List<String> findAllPhoneNumbersByTenantId(@Param("tenantId") UUID tenantId);

    /**
     * Used ONLY during first-login account provisioning, before we know
     * which tenant a phone number belongs to. Deliberately NOT tenant
     * scoped - see AuthService.provisionAccountForPhone for the documented
     * limitation this implies (a guardian phone matching students in more
     * than one tenant will resolve to the first match found).
     */
    Optional<Student> findFirstByGuardianPhoneAndActiveTrue(String guardianPhone);

    Optional<Student> findFirstByPhoneNumberAndActiveTrue(String phoneNumber);
}

package com.teacheros.modules.audit;

import com.teacheros.common.tenant.CurrentTenantProvider;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Set;
import java.util.UUID;
import java.util.regex.Pattern;

/**
 * Writes who-did-what to audit_log. Tenant and actor always come from the
 * JWT via CurrentTenantProvider. The row is written in the CALLER'S
 * transaction, so an audit row exists exactly when the action committed.
 *
 * Metadata is sanitised before storage: keys that look like secrets (otp,
 * code, token, password, secret, signature) are dropped, and any phone
 * number - by key name or by value shape - is masked to its last 4 digits.
 */
@Service
public class AuditService {

    private static final Pattern PHONE_VALUE = Pattern.compile("^\\+?[0-9]{10,15}$");
    private static final Set<String> BLOCKED_KEY_FRAGMENTS = Set.of("otp", "code", "token", "password", "secret",
            "signature");

    private final AuditLogRepository auditLogRepository;
    private final CurrentTenantProvider currentTenantProvider;

    public AuditService(AuditLogRepository auditLogRepository, CurrentTenantProvider currentTenantProvider) {
        this.auditLogRepository = auditLogRepository;
        this.currentTenantProvider = currentTenantProvider;
    }

    @Transactional
    public void record(String action, String resourceType, UUID resourceId, Map<String, Object> metadata) {
        auditLogRepository.save(new AuditLog(
                currentTenantProvider.getCurrentTenantId(),
                currentTenantProvider.getCurrentUserId(),
                action, resourceType, resourceId, sanitize(metadata)));
    }

    /**
     * Builds a metadata map from key/value pairs; unlike Map.of, null values are
     * allowed.
     */
    public static Map<String, Object> meta(Object... keyValues) {
        Map<String, Object> map = new LinkedHashMap<>();
        for (int i = 0; i + 1 < keyValues.length; i += 2) {
            map.put(String.valueOf(keyValues[i]), keyValues[i + 1]);
        }
        return map;
    }

    /** Keeps only the last 4 characters, e.g. +919876543210 -> *********3210. */
    public static String maskPhone(String phone) {
        if (phone == null || phone.length() <= 4) {
            return "****";
        }
        return "*".repeat(phone.length() - 4) + phone.substring(phone.length() - 4);
    }

    static Map<String, Object> sanitize(Map<String, Object> metadata) {
        Map<String, Object> clean = new LinkedHashMap<>();
        if (metadata == null) {
            return clean;
        }
        for (Map.Entry<String, Object> e : metadata.entrySet()) {
            String key = e.getKey();
            if (key == null) {
                continue;
            }
            String lower = key.toLowerCase();
            if (BLOCKED_KEY_FRAGMENTS.stream().anyMatch(lower::contains)) {
                continue;
            }
            Object value = e.getValue();
            if (value instanceof String s) {
                clean.put(key, (lower.contains("phone") || PHONE_VALUE.matcher(s).matches()) ? maskPhone(s) : s);
            } else if (value == null || value instanceof Number || value instanceof Boolean) {
                clean.put(key, value);
            } else {
                clean.put(key, String.valueOf(value)); // UUIDs, dates, enums -> plain strings
            }
        }
        return clean;
    }
}

package com.teacheros.modules.permission;

import com.teacheros.common.tenant.CurrentTenantProvider;
import com.teacheros.modules.user.Role;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.UUID;

/**
 * Registered as the bean "perm" so controllers can write
 * {@code @PreAuthorize("@perm.has('FEES_EDIT')")} instead of a long role list.
 *
 * Tenant and role always come from the authenticated principal
 * (CurrentTenantProvider), never from the request.
 *
 * Fails closed: no row for (tenant, role, permission) means denied. OWNER is
 * always allowed so an edited permission matrix can never lock the owner out.
 *
 * A misspelled permission name throws IllegalArgumentException (a 500, not a
 * silent allow). PreAuthorizePermissionNamesTest catches typos at build time.
 */
@Service("perm")
public class PermissionService {

    private final RolePermissionRepository rolePermissionRepository;
    private final CurrentTenantProvider currentTenantProvider;

    public PermissionService(RolePermissionRepository rolePermissionRepository,
            CurrentTenantProvider currentTenantProvider) {
        this.rolePermissionRepository = rolePermissionRepository;
        this.currentTenantProvider = currentTenantProvider;
    }

    // Single String-arg method (no enum overload) so SpEL resolution is
    // unambiguous.
    public boolean has(String permissionName) {
        Permission permission = Permission.valueOf(permissionName);
        String role = currentTenantProvider.getCurrentRole();

        if (Role.OWNER.name().equals(role)) {
            return true;
        }
        return rolePermissionRepository.existsByTenantIdAndRoleAndPermission(
                currentTenantProvider.getCurrentTenantId(), role, permission.name());
    }

    /**

package com.teacheros.modules.feestructure;

import com.teacheros.common.exception.ApiException;
import com.teacheros.common.exception.ResourceNotFoundException;
import com.teacheros.common.tenant.CurrentTenantProvider;

@Service
public class FeeStructureService {

    private final FeeStructureRepository structureRepository;
    private final FeeStructureItemRepository itemRepository;
    private final FeeHeadRepository headRepository;
    private final InstalmentPlanRepository planRepository;
    private final BatchRepository batchRepository;
    private final FeeAuditWriter audit;
    private final CurrentTenantProvider currentTenantProvider;

    public FeeStructureService(FeeStructureRepository structureRepository, FeeStructureItemRepository itemRepository,
            FeeHeadRepository headRepository, InstalmentPlanRepository planRepository,
            BatchRepository batchRepository, FeeAuditWriter audit, CurrentTenantProvider currentTenantProvider) {
        this.structureRepository = structureRepository;
        this.itemRepository = itemRepository;
        this.headRepository = headRepository;
        this.planRepository = planRepository;
        this.batchRepository = batchRepository;
        this.audit = audit;
        this.currentTenantProvider = currentTenantProvider;
    }

    public List<FeeStructureResponse> list() {
        UUID tenantId = currentTenantProvider.getCurrentTenantId();
        List<FeeStructure> structures = structureRepository.findByTenantIdOrderByAcademicYearDescNameAsc(tenantId);
        if (structures.isEmpty()) {
            return List.of();
        }
        Map<UUID, List<FeeStructureItem>> items = itemRepository
                .findByTenantIdAndStructureIdIn(tenantId, structures.stream().map(FeeStructure::getId).toList())
                .stream().collect(Collectors.groupingBy(FeeStructureItem::getStructureId));
        Map<UUID, FeeHead> heads = headRepository.findByTenantIdOrderBySortOrderAscNameAsc(tenantId).stream()
                .collect(Collectors.toMap(FeeHead::getId, Function.identity()));
        Map<UUID, InstalmentPlan> plans = planRepository.findByTenantIdOrderByNameAsc(tenantId).stream()
                .collect(Collectors.toMap(InstalmentPlan::getId, Function.identity()));
        return structures.stream()
                .map(s -> toResponse(s, items.getOrDefault(s.getId(), List.of()), heads, plans.get(s.getPlanId())))
                .toList();
    }

    @Transactional
    public FeeStructureResponse create(CreateFeeStructureRequest request) {
        UUID tenantId = currentTenantProvider.getCurrentTenantId();
        String name = request.name().trim();

        InstalmentPlan plan = planRepository.findByIdAndTenantId(request.planId(), tenantId)
                .orElseThrow(() -> new ResourceNotFoundException("Instalment plan", request.planId()));
        if (request.batchId() != null) {
            batchRepository.findByIdAndTenantId(request.batchId(), tenantId)
                    .orElseThrow(() -> new ResourceNotFoundException("Batch", request.batchId()));
        }
        if (structureRepository.existsByTenantIdAndNameIgnoreCaseAndAcademicYear(tenantId, name, request.academicYear())) {
            throw new ApiException(HttpStatus.CONFLICT, "STRUCTURE_EXISTS",
                    "A structure named \"" + name + "\" already exists for " + request.academicYear() + ".");
        }

        Set<UUID> headIds = new HashSet<>();
        for (StructureItemInput item : request.items()) {
            if (!headIds.add(item.headId())) {
                throw new ApiException(HttpStatus.BAD_REQUEST, "DUPLICATE_HEAD", "Each fee head can appear only once.");
            }
        }
        Map<UUID, FeeHead> heads = headRepository.findByTenantIdAndIdIn(tenantId, headIds).stream()
                .collect(Collectors.toMap(FeeHead::getId, Function.identity()));
        for (UUID headId : headIds) {
            FeeHead head = heads.get(headId);
            if (head == null) {
                throw new ResourceNotFoundException("Fee head", headId);
            }
            if (!head.isActive()) {
                throw new ApiException(HttpStatus.BAD_REQUEST, "HEAD_INACTIVE", "Fee head \"" + head.getName() + "\" is inactive.");
            }
        }

        String grade = request.grade() == null || request.grade().isBlank() ? null : request.grade().trim();
        FeeStructure structure = structureRepository.save(
                new FeeStructure(tenantId, name, request.academicYear(), grade, request.batchId(), plan.getId()));
        List<FeeStructureItem> items = itemRepository.saveAll(request.items().stream()
                .map(i -> new FeeStructureItem(tenantId, structure.getId(), i.headId(), FeeCalculator.scale(i.amount())))
                .toList());

        audit.record("FEE_STRUCTURE_CREATED", "FEE_STRUCTURE", structure.getId(),
                Map.of("name", name, "academicYear", request.academicYear(), "items", items.size()));
        return toResponse(structure, items, heads, plan);
    }

    @Transactional
    public FeeStructureResponse update(UUID id, UpdateFeeStructureRequest request) {
        UUID tenantId = currentTenantProvider.getCurrentTenantId();
        FeeStructure structure = structureRepository.findByIdAndTenantId(id, tenantId)
                .orElseThrow(() -> new ResourceNotFoundException("Fee structure", id));
        structure.setName(request.name().trim());
        structure.setActive(request.active());
        structureRepository.save(structure);

        Map<UUID, FeeHead> heads = headRepository.findByTenantIdOrderBySortOrderAscNameAsc(tenantId).stream()
                .collect(Collectors.toMap(FeeHead::getId, Function.identity()));
        InstalmentPlan plan = planRepository.findByIdAndTenantId(structure.getPlanId(), tenantId).orElse(null);
        return toResponse(structure, itemRepository.findByTenantIdAndStructureId(tenantId, id), heads, plan);
    }

    private FeeStructureResponse toResponse(FeeStructure s, List<FeeStructureItem> items, Map<UUID, FeeHead> heads,
            InstalmentPlan plan) {
        List<StructureItemResponse> rows = items.stream()
                .sorted(Comparator.comparingInt(i -> heads.containsKey(i.getHeadId()) ? heads.get(i.getHeadId()).getSortOrder() : 0))
                .map(i -> new StructureItemResponse(i.getHeadId(),
                        heads.containsKey(i.getHeadId()) ? heads.get(i.getHeadId()).getName() : "Unknown", i.getAmount()))
                .toList();
        BigDecimal total = items.stream().map(FeeStructureItem::getAmount).reduce(BigDecimal.ZERO, BigDecimal::add);
        return new FeeStructureResponse(s.getId(), s.getName(), s.getAcademicYear(), s.getGrade(), s.getBatchId(),
                s.getPlanId(), plan == null ? null : plan.getName(), s.isActive(), rows, FeeCalculator.scale(total));
    }
}

package com.teacheros.modules.imports;

import com.teacheros.common.exception.ApiException;
import com.teacheros.common.tenant.CurrentTenantProvider;
import com.teacheros.modules.batch.Batch;
import com.teacheros.modules.batch.BatchRepository;
import com.teacheros.modules.imports.TabularFileReader.ParsedFile;
import com.teacheros.modules.imports.TabularFileReader.RawRow;
import com.teacheros.modules.imports.dto.ImportDtos.ImportResult;
import com.teacheros.modules.imports.dto.ImportDtos.RowError;
import com.teacheros.modules.student.Student;
import com.teacheros.modules.student.StudentRepository;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.multipart.MultipartFile;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Locale;
import java.util.Map;
import java.util.Set;

/**
 * Bulk student import from a .csv/.xlsx file. Same validation rules as
 * StudentDtos.CreateStudentRequest, plus the checks a single-record create
 * gets for free from the database but a 500-row file needs reported per row
 * (duplicate phone numbers, unknown batch names).
 *
 * Tenant scoping: the tenant comes only from CurrentTenantProvider and
 * batchName is resolved only against that tenant's own batches, so a file
 * can never attach a student to another tenant's batch or create data in
 * another tenant.
 */
@Service
public class StudentImportService {

    private static final long MAX_FILE_BYTES = 5L * 1024 * 1024;
    private static final Pattern PHONE = Pattern.compile("^\\+?[0-9]{10,15}$");
    private static final Pattern EMAIL = Pattern.compile("^[^@\\s]+@[^@\\s]+\\.[^@\\s]+$");
    private static final String EXPECTED_COLUMNS = "fullName, grade, guardianPhone, phoneNumber, guardianEmail, batchName";

    // Header text is lower-cased with everything but letters/digits removed
    // before lookup, so "Full Name", "full_name" and "FULLNAME" all match.
    private static final Map<String, String> HEADER_ALIASES = Map.ofEntries(
            Map.entry("fullname", "fullName"),
            Map.entry("name", "fullName"),
            Map.entry("studentname", "fullName"),
            Map.entry("grade", "grade"),
            Map.entry("class", "grade"),
            Map.entry("guardianphone", "guardianPhone"),
            Map.entry("guardianmobile", "guardianPhone"),
            Map.entry("parentphone", "guardianPhone"),
            Map.entry("parentmobile", "guardianPhone"),
            Map.entry("phonenumber", "phoneNumber"),
            Map.entry("studentphone", "phoneNumber"),
            Map.entry("studentphonenumber", "phoneNumber"),
            Map.entry("guardianemail", "guardianEmail"),
            Map.entry("parentemail", "guardianEmail"),
            Map.entry("batch", "batchName"),
            Map.entry("batchname", "batchName"));

    private final StudentRepository studentRepository;
    private final BatchRepository batchRepository;
    private final CurrentTenantProvider currentTenantProvider;
    private final int maxRows;

    public StudentImportService(
            StudentRepository studentRepository,
            BatchRepository batchRepository,
            CurrentTenantProvider currentTenantProvider,
            @Value("${app.imports.max-rows:2000}") int maxRows) {
        this.studentRepository = studentRepository;
        this.batchRepository = batchRepository;
        this.currentTenantProvider = currentTenantProvider;
        this.maxRows = maxRows;
    }

    @Transactional
    public ImportResult importStudents(MultipartFile file, boolean dryRun, boolean skipInvalidRows) {
        UUID tenantId = currentTenantProvider.getCurrentTenantId();

        if (file == null || file.isEmpty()) {
            throw new ApiException(HttpStatus.BAD_REQUEST, "FILE_REQUIRED", "Choose a file to import.");
        }
        if (file.getSize() > MAX_FILE_BYTES) {
            throw new ApiException(HttpStatus.BAD_REQUEST, "FILE_TOO_LARGE",
                    "That file is too large. Maximum size is " + (MAX_FILE_BYTES / (1024 * 1024)) + "MB.");
        }

        ParsedFile parsed = TabularFileReader.read(file, maxRows);
        Map<String, Integer> cols = mapColumns(parsed.headers());
        if (parsed.rows().isEmpty()) {
            throw new ApiException(HttpStatus.BAD_REQUEST, "IMPORT_NO_ROWS",
                    "The file has a header but no student rows.");
        }

        // Everything below is looked up once, up front - not once per row.
        Map<String, List<Batch>> batchesByName = new HashMap<>();
        for (Batch b : batchRepository.findAllByTenantIdAndActiveTrue(tenantId)) {
            batchesByName.computeIfAbsent(normalizeName(b.getName()), k -> new ArrayList<>()).add(b);
        }
        // All phone numbers, including inactive students' - the unique index
        // uq_students_tenant_phone covers every row, not just active ones.
        Set<String> dbPhones = new HashSet<>(studentRepository.findAllPhoneNumbersByTenantId(tenantId));
        Set<String> dbStudentKeys = new HashSet<>();
        for (Student s : studentRepository.findAllByTenantIdAndActiveTrue(tenantId)) {
            dbStudentKeys.add(studentKey(s.getFullName(), s.getGuardianPhone()));
        }

        Map<String, Integer> filePhoneLine = new HashMap<>();
        Map<String, Integer> fileKeyLine = new HashMap<>();
        List<Student> valid = new ArrayList<>();
        List<RowError> errors = new ArrayList<>();

        for (RawRow row : parsed.rows()) {
            List<String> messages = new ArrayList<>();

            String fullName = cell(row, cols, "fullName");
            String grade = cell(row, cols, "grade");
            String guardianPhone = normalizePhone(cell(row, cols, "guardianPhone"));
            String studentPhone = normalizePhone(cell(row, cols, "phoneNumber"));
            String guardianEmail = cell(row, cols, "guardianEmail");
            String batchName = cell(row, cols, "batchName");

            if (fullName.isEmpty()) {
                messages.add("Full name is required.");
            } else if (fullName.length() > 255) {
                messages.add("Full name is too long (max 255 characters).");
            }
            if (grade.length() > 50) {
                messages.add("Grade is too long (max 50 characters).");
            }

            boolean guardianPhoneValid = false;
            if (guardianPhone.isEmpty()) {
                messages.add("Guardian phone is required.");
            } else if (!PHONE.matcher(guardianPhone).matches()) {
                messages.add("Guardian phone must be 10-15 digits, optionally starting with +.");
            } else {
                guardianPhoneValid = true;
            }

            if (!studentPhone.isEmpty()) {
                if (!PHONE.matcher(studentPhone).matches()) {
                    messages.add("Student phone must be 10-15 digits, optionally starting with +.");
                } else if (dbPhones.contains(studentPhone)) {
                    messages.add("Student phone " + studentPhone + " is already used by another student.");
                } else if (filePhoneLine.containsKey(studentPhone)) {
                    messages.add("Student phone " + studentPhone + " is already used on line "
                            + filePhoneLine.get(studentPhone) + ".");
                }
            }

            if (!guardianEmail.isEmpty() && (guardianEmail.length() > 255 || !EMAIL.matcher(guardianEmail).matches())) {
                messages.add("Guardian email is not a valid email address.");
            }

            UUID batchId = null;
            if (!batchName.isEmpty()) {
                List<Batch> matches = batchesByName.getOrDefault(normalizeName(batchName), List.of());
                if (matches.isEmpty()) {
                    messages.add(
                            "Batch \"" + batchName + "\" was not found. Check the spelling or create the batch first.");
                } else if (matches.size() > 1) {
                    messages.add("More than one batch is named \"" + batchName + "\". Rename one so names are unique.");
                } else {
                    batchId = matches.get(0).getId();
                }
            }

            String key = null;
            if (!fullName.isEmpty() && guardianPhoneValid) {
                key = studentKey(fullName, guardianPhone);
                if (dbStudentKeys.contains(key)) {
                    messages.add("A student named \"" + fullName + "\" with this guardian phone already exists.");
                } else if (fileKeyLine.containsKey(key)) {
                    messages.add("Duplicate of line " + fileKeyLine.get(key) + " (same name and guardian phone).");
                }
            }

            if (messages.isEmpty()) {
                Student student = new Student(tenantId, batchId, fullName, grade.isEmpty() ? null : grade,
                        guardianPhone);
                student.setPhoneNumber(studentPhone.isEmpty() ? null : studentPhone);
                student.setGuardianEmail(guardianEmail.isEmpty() ? null : guardianEmail);
                valid.add(student);
                // Only valid rows are remembered, so one bad row doesn't make
                // every later row that resembles it look like a duplicate.
                if (!studentPhone.isEmpty()) {
                    filePhoneLine.put(studentPhone, row.lineNumber());
                }
                fileKeyLine.put(key, row.lineNumber());
            } else {
                errors.add(new RowError(row.lineNumber(), messages));
            }
        }

        int imported = 0;
        boolean blockedByErrors = !errors.isEmpty() && !skipInvalidRows;
        if (!dryRun && !blockedByErrors && !valid.isEmpty()) {
            try {
                // Flushed here so a unique-index clash with a concurrent
                // request surfaces inside this try, not at commit.
                studentRepository.saveAllAndFlush(valid);
            } catch (DataIntegrityViolationException e) {
                throw new ApiException(HttpStatus.CONFLICT, "IMPORT_CONFLICT",
                        "Someone else changed the student list while this file was importing. Nothing was imported - please try again.");
            }
            imported = valid.size();
        }

        return new ImportResult(dryRun, parsed.rows().size(), valid.size(), imported, errors.size(), errors);
    }

    private Map<String, Integer> mapColumns(List<String> headers) {
        Map<String, Integer> cols = new HashMap<>();
        for (int i = 0; i < headers.size(); i++) {
            String canonical = HEADER_ALIASES.get(headers.get(i).toLowerCase(Locale.ROOT).replaceAll("[^a-z0-9]", ""));
            if (canonical != null) {
                cols.putIfAbsent(canonical, i);
            }
        }
        List<String> missing = new ArrayList<>();
        for (String required : List.of("fullName", "guardianPhone")) {
            if (!cols.containsKey(required)) {
                missing.add(required);
            }
        }
        if (!missing.isEmpty()) {
            throw new ApiException(HttpStatus.BAD_REQUEST, "MISSING_COLUMNS",
                    "The file is missing required column(s): " + String.join(", ", missing)
                            + ". Expected columns: " + EXPECTED_COLUMNS + ".");
        }
        return cols;
    }

    private static String cell(RawRow row, Map<String, Integer> cols, String key) {
        Integer idx = cols.get(key);
        if (idx == null || idx >= row.cells().size()) {
            return "";
        }
        String value = row.cells().get(idx);
        return value == null ? "" : value.trim();
    }

    private static String normalizePhone(String raw) {
        return raw.replaceAll("[\\s\\-().]", "");
    }

    private static String normalizeName(String name) {
        return name.trim().replaceAll("\\s+", " ").toLowerCase(Locale.ROOT);
    }

    private static String studentKey(String fullName, String guardianPhone) {
        return normalizeName(fullName) + "|" + guardianPhone;
    }
}
